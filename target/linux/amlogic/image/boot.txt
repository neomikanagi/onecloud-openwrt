# OpenWrt boot script for Thunder OneCloud (Amlogic S805 / meson8b).
#
# Compiled to boot.scr and placed on the FAT boot partition. Kept to the
# subset of U-Boot hush that the stock OneCloud u-boot understands:
# setenv / test / if-then-fi / fatload / bootm only. No case, no for.

setenv condev "console=ttyAML0,115200n8 no_console_suspend consoleblank=0"
setenv kernel_loadaddr "0x00208000"
setenv dtb_loadaddr "0x21800000"
setenv bootargs_common "rootfstype=ext4 rootwait ro ${condev} fsck.repair=yes net.ifnames=0"

# 1) eMMC (mmc 1) - the normal path
if fatload mmc 1 ${kernel_loadaddr} uImage; then
	if fatload mmc 1 ${dtb_loadaddr} dtb; then
		echo "OneCloud: booting from eMMC"
		setenv bootargs "root=/dev/mmcblk1p2 ${bootargs_common}"
		bootm ${kernel_loadaddr} - ${dtb_loadaddr}
	fi
fi

# 2) SD card (mmc 0)
if fatload mmc 0 ${kernel_loadaddr} uImage; then
	if fatload mmc 0 ${dtb_loadaddr} dtb; then
		echo "OneCloud: booting from SD"
		setenv bootargs "root=/dev/mmcblk0p2 ${bootargs_common}"
		bootm ${kernel_loadaddr} - ${dtb_loadaddr}
	fi
fi

# 3) USB - used to boot the installer when eMMC has no system yet
usb start
if fatload usb 0 ${kernel_loadaddr} uImage; then
	if fatload usb 0 ${dtb_loadaddr} dtb; then
		echo "OneCloud: booting from USB"
		setenv bootargs "root=/dev/sda2 ${bootargs_common}"
		bootm ${kernel_loadaddr} - ${dtb_loadaddr}
	fi
fi

echo "OneCloud: no bootable kernel found on eMMC, SD or USB"
